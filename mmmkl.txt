local Util = {}

Util.RandomString = function(length)
	length = length or math.random(32, 256)
	local str = ""
	for _ = 1, length do
		str ..= string.char(math.random(32, 126))
	end
	return str
end

Util.Instance = function(cl, p)
	local i = Instance.new(cl)
	i.Name = Util.RandomString()
	i.Parent = p
	return i
end

Util.LinkDestroyI2C = function(a, b)
	a.Destroying:Once(function()
		b:Disconnect()
	end)
end

Util.LoopedHSV = function(h, s, v)
	h %= 1
	s = math.clamp(s, 0, 1)
	v = math.clamp(v, 0, 1)
	return Color3.fromHSV(h, s, v)
end

Util.Notify = function(text, title)
	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = title or "UI",
			Text = text,
			Duration = 3,
		})
	end)
end

Util.ForceTextSize = function(inst)
	if inst:IsA("TextLabel") then
		local original = inst.TextSize
		local conn = RunService.PreRender:Connect(function()
			local size = GuiService.PreferredTextSize.Name
			if size == "Medium" then inst.TextSize = original
			elseif size == "Large" then inst.TextSize = original - 4
			elseif size == "Larger" then inst.TextSize = original - 9
			elseif size == "Largest" then inst.TextSize = original - 14
			end
		end)
		Util.LinkDestroyI2C(inst, conn)
	end
end

Util.MakeHat = function(color, dur)
	dur = dur or 8
	local pivot = Util.Instance("Frame")
	pivot.AnchorPoint = Vector2.new(0.5, 0.5)
	pivot.Position = UDim2.new(0.5, 0, 0.5, 0)
	pivot.Size = UDim2.new(1, 0, 1, 0)
	pivot.BackgroundTransparency = 1
	pivot.BorderSizePixel = 0

	local grey = math.max(color.R, color.G, color.B) * 0.5

	local triW = 0.42
	local triH = 0.52
	local thickness = 0.025

	-- 左边斜边
	local leftBar = Util.Instance("Frame", pivot)
	leftBar.AnchorPoint = Vector2.new(0.5, 0.5)
	leftBar.Position = UDim2.new(0.5 - triW * 0.25, 0, 0.78 - triH * 0.5, 0)
	leftBar.Size = UDim2.new(thickness, 0, math.sqrt(triW * triW + triH * triH) * 0.98, 0)
	leftBar.BackgroundColor3 = Color3.new(grey, grey, grey)
	leftBar.BorderSizePixel = 0
	leftBar.Rotation = -math.deg(math.atan2(triH, triW * 0.5))

	-- 右边斜边
	local rightBar = Util.Instance("Frame", pivot)
	rightBar.AnchorPoint = Vector2.new(0.5, 0.5)
	rightBar.Position = UDim2.new(0.5 + triW * 0.25, 0, 0.78 - triH * 0.5, 0)
	rightBar.Size = UDim2.new(thickness, 0, math.sqrt(triW * triW + triH * triH) * 0.98, 0)
	rightBar.BackgroundColor3 = Color3.new(grey, grey, grey)
	rightBar.BorderSizePixel = 0
	rightBar.Rotation = math.deg(math.atan2(triH, triW * 0.5))

	-- 底边
	local bottomBar = Util.Instance("Frame", pivot)
	bottomBar.AnchorPoint = Vector2.new(0.5, 0.5)
	bottomBar.Position = UDim2.new(0.5, 0, 0.78, 0)
	bottomBar.Size = UDim2.new(triW, 0, thickness, 0)
	bottomBar.BackgroundColor3 = Color3.new(grey, grey, grey)

	-- 内部填充

	innerFill.AnchorPoint = Vector2.new(0.5, 0.5)
	innerFill.Position = UDim2.new(0.5, 0, 0.78 - triH * 0.33, 0)

	return pivot

Util.MakeText = function(text)
	text = text:upper()
	local ls = "BCDEIKOPQRSTUVYZ\\_`N^MJ"
	local lt = ""
	lt ..= "DDDDD@DFF@@@@@FSFSFSF@@@@@@@@@@@@@@EMMUKKTDD@@@@@DEEEEEDDBBBBBD@@@@@@@@@@@@@@@@@@@DE@@@G@@@@@@@@@DABBDEEI"
	lt ..= "GJJJJJGDVDDDDSGJAGIISRAAGAARWJJSAAASIIRAARGIIRJJGSAABBDDGJJGJJGGJJHAAG"
	lt ..= "@DD@DD@@DD@DE@@BDEDB@@@G@G@@@EDBDE@GJABD@D@@@@@@@"
	lt ..= "DFJSJJJRJJRJJRGJIIIJGRJJJJJRSIIRIISSIIRIIIHIILJJHJJJSJJJGDDDDDGAAAAAJHJKMOMKJIIIIIISJQNJJJJJJPNLJJGJJJJJGRJJRIIIGJJJNGCRJJRMKJGJIGAJGSDDDDDDJJJJJJGJJJJJFDJJJNQJJJJFDFJJJJFDDDDSABDEIS"
	local pixs = {}
	local totalsize = 0
	for i = 0, 6 do
		local row = {}
		for j = 1, text:len() do
			local c = text:byte(j, j)
			local w = 0
			if c ~= 0x20 then
				local k = (c - 33) * 7 + i
				w = lt:byte(k + 1, k + 1)
				if w ~= nil and w > 64 then
					w = ls:byte(w - 64, w - 64) - 65
				else
					w = 0
				end
			end
			for k = 0, 5 do
				local h = (j - 1) * 6 + (4 - k)
				totalsize = math.max(totalsize, h + 1)
				local p = (w // math.pow(2, k)) % 2
				if p > 0 then
					row[h] = true
				end
			end
		end
		pixs[i] = row
	end
	local pivot = Util.Instance("Frame")
	pivot.AnchorPoint = Vector2.new(0.5, 0.5)
	pivot.Position = UDim2.new(0.5, 0, 0.5, 0)
	pivot.Size = UDim2.new(0, totalsize, 0, 7)
	pivot.BackgroundTransparency = 1
	pivot.BorderSizePixel = 0
	for i = 0, 6 do
		local row = pixs[i]
		local olde = nil
		local olds = 0
		for j = 0, totalsize - 1 do
			local pix = row[j]
			if pix == true then
				if olde ~= nil then
					olds += 1
					olde.Size = UDim2.new(olds / totalsize, 0, 1 / 7, 0)
				else
					olde = Util.Instance("Frame", pivot)
					olde.AnchorPoint = Vector2.new(0, 0)
					olde.Position = UDim2.new(j / totalsize, 0, i / 7, 0)
					olde.Size = UDim2.new(1 / totalsize, 0, 1 / 7, 0)
					olde.BackgroundTransparency = 1
					olde.BorderSizePixel = 0
					olds = 1
				end
			else
				olde = nil
				olds = 0
			end
		end
	end
	return pivot
end

		v.BackgroundTransparency = tran
	end

local StylizedObjs = {}
local function Stylize(obj, options)
	options = options or {}
	Util.Instance("UICorner", obj).CornerRadius = UDim.new(0, 5)
	local Out = Util.Instance("UIStroke", obj)
	Out.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	Out.Color = Color3.new(1, 1, 1)
	Out.LineJoinMode = Enum.LineJoinMode.Round
	Out.Thickness = 1
	Out.Transparency = 0
	Out.Enabled = true
	obj.BackgroundColor3 = Color3.new(0, 0, 0)
	table.insert(StylizedObjs, { obj = obj, Out = Out })
end

local function GetUIColor(t)
	return Util.LoopedHSV(t / 10, 0.8, 1)
end
local function GetUIBGColor(t)
	return Color3.new(0, 0, 0)
end
local function UpdateGrads(t)
	local c = GetUIColor(t)
	local bgc = GetUIBGColor(t)
	for _, grad in StylizedObjs do
		grad.Out.Color = c
		grad.obj.BackgroundColor3 = bgc
	end
end

local _funcrefreshes = {}
local function AddToRenderStep(func, linkto)
	table.insert(_funcrefreshes, func)
	if linkto then
		linkto.Destroying:Connect(function()
			local i = table.find(_funcrefreshes, func)
			if i then table.remove(_funcrefreshes, i) end
		end)
	end
	return func
end

function UI.CreateButton(parent, text, size)
	local margin = 5
	local Container = Util.Instance("Frame", parent)
	Container.AnchorPoint = Vector2.new(0.5, 0)
	Container.Size = UDim2.new(1, 0, 0, 65536)
	Container.BackgroundTransparency = 1

	ButtonText.TextYAlignment = Enum.TextYAlignment.Center

	return Button, ButtonText

function UI.CreateSlider(parent, text, min, max, default, callback)
	min = min or 0
	max = max or 100
	default = default or min
	callback = callback or function() end

	local margin = 5
	local Container = Util.Instance("Frame", parent)
	Container.AnchorPoint = Vector2.new(0.5, 0)
	Container.Size = UDim2.new(1, 0, 0, 52)
	Container.BackgroundTransparency = 1
	Container.LayoutOrder = #parent:GetChildren()

	local Back = Util.Instance("Frame", Container)
	Back.AnchorPoint = Vector2.new(0, 0)
	Back.Position = UDim2.new(0, margin, 0, margin // 2)
	Back.Size = UDim2.new(1, margin * -2, 1, -margin)
	Back.BackgroundColor3 = Color3.new(1, 1, 1)
	Back.BorderSizePixel = 0
	Stylize(Back)

	local Label = Util.Instance("TextLabel", Back)
	Label.Position = UDim2.new(0, 10, 0, 6)
	Label.Size = UDim2.new(1, -70, 0, 18)
	Label.BackgroundTransparency = 1
	Label.Font = Enum.Font.Code
	Label.TextColor3 = Color3.new(1, 1, 1)
	Label.TextSize = 16
	Label.TextXAlignment = Enum.TextXAlignment.Left
	Label.Text = text
	RegisterTextLabel(Label)

	local ValueLabel = Util.Instance("TextLabel", Back)
	ValueLabel.AnchorPoint = Vector2.new(1, 0)
	ValueLabel.Position = UDim2.new(1, -10, 0, 6)
	ValueLabel.Size = UDim2.new(0, 60, 0, 18)
	ValueLabel.BackgroundTransparency = 1
	ValueLabel.Font = Enum.Font.Code
	ValueLabel.TextColor3 = Color3.new(1, 1, 1)
	ValueLabel.TextSize = 16
	ValueLabel.TextXAlignment = Enum.TextXAlignment.Right
	ValueLabel.Text = tostring(default)
	RegisterTextLabel(ValueLabel)

	local Bar = Util.Instance("Frame", Back)
	Bar.Position = UDim2.new(0, 10, 0, 32)
	Bar.Size = UDim2.new(1, -20, 0, 10)
	Bar.BackgroundColor3 = Color3.new(1, 1, 1)
	Bar.BackgroundTransparency = 0.7
	Bar.BorderSizePixel = 0
	Util.Instance("UICorner", Bar).CornerRadius = UDim.new(1, 0)

	local pct = (default - min) / (max - min)
	local Fill = Util.Instance("Frame", Bar)
	Fill.Size = UDim2.new(pct, 0, 1, 0)
	Fill.BackgroundColor3 = UITextColor.Value
	Fill.BorderSizePixel = 0
	Util.Instance("UICorner", Fill).CornerRadius = UDim.new(1, 0)
	Util.LinkDestroyI2C(Fill, UITextColor.Changed:Connect(function(val)
		Fill.BackgroundColor3 = val
	end))

	local Knob = Util.Instance("Frame", Bar)
	Knob.AnchorPoint = Vector2.new(0.5, 0.5)
	Knob.Position = UDim2.new(pct, 0, 0.5, 0)
	Knob.Size = UDim2.new(0, 14, 0, 14)
	Knob.BackgroundColor3 = Color3.new(1, 1, 1)
	Knob.BorderSizePixel = 0
	Util.Instance("UICorner", Knob).CornerRadius = UDim.new(1, 0)
	Stylize(Knob)

	local dragging = false
	local currentValue = default
	local function setFromX(x)
		local ax = Bar.AbsolutePosition.X
		local aw = math.max(1, Bar.AbsoluteSize.X)
		local t = math.clamp((x - ax) / aw, 0, 1)
		local v = math.floor(min + (max - min) * t + 0.5)
		local p = (v - min) / (max - min)
		Fill.Size = UDim2.new(p, 0, 1, 0)
		Knob.Position = UDim2.new(p, 0, 0.5, 0)
		ValueLabel.Text = tostring(v)
		currentValue = v
		pcall(callback, v)
	end

	local Hit = Util.Instance("TextButton", Back)
	Hit.Position = UDim2.new(0, 0, 0, 22)
	Hit.Size = UDim2.new(1, 0, 0, 30)
	Hit.BackgroundTransparency = 1
	Hit.Text = ""
	Hit.ZIndex = 5
	Hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	return {
		Set = function(v)
			local p = (math.clamp(v, min, max) - min) / (max - min)
			Fill.Size = UDim2.new(p, 0, 1, 0)
			Knob.Position = UDim2.new(p, 0, 0.5, 0)
			ValueLabel.Text = tostring(v)
			currentValue = v
			pcall(callback, v)
		end,
		Get = function() return currentValue end,
	}
end

function UI.CreateDropdown(parent, text, options, callback)
	options = options or {}
	callback = callback or function() end

	local margin = 5

	Back.Size = UDim2.new(1, margin * -2, 1, -margin)
	Back.BackgroundColor3 = Color3.new(1, 1, 1)
	Back.BorderSizePixel = 0
	Back.Text = ""
	Back.AutoButtonColor = false
	Stylize(Back)

	local Label = Util.Instance("TextLabel", Back)
	Label.AnchorPoint = Vector2.new(0, 0.5)
	Label.Position = UDim2.new(0, 10, 0.5, 0)

	Label.TextColor3 = Color3.new(1, 1, 1)
	Label.TextSize = 18
	Label.TextXAlignment = Enum.TextXAlignment.Left
	Label.Text = text
	RegisterTextLabel(Label)

	local Arrow = Util.Instance("TextLabel", Back)
	Arrow.AnchorPoint = Vector2.new(1, 0.5)
	Arrow.Position = UDim2.new(1, -8, 0.5, 0)
	Arrow.Size = UDim2.new(0, 20, 1, 0)
	Arrow.BackgroundTransparency = 1
	Arrow.Font = Enum.Font.Code
	Arrow.TextColor3 = Color3.new(1, 1, 1)
	Arrow.TextSize = 16
	Arrow.Text = "▼"
	RegisterTextLabel(Arrow)

	local List = Util.Instance("Frame", Container)
	List.Position = UDim2.new(0, margin, 0, 36)
	List.Size = UDim2.new(1, margin * -2, 0, 0)
	List.BackgroundTransparency = 1
	List.ClipsDescendants = true

	local Layout = Util.Instance("UIListLayout", List)
	Layout.FillDirection = Enum.FillDirection.Vertical
	Layout.SortOrder = Enum.SortOrder.LayoutOrder
	Layout.Padding = UDim.new(0, 3)

	local optionButtons = {}
	local isOpen = false
	local currentValue = nil

	local function refreshOptions(newOptions)
		for _, b in ipairs(optionButtons) do
			b:Destroy()
		end
		optionButtons = {}
		options = newOptions or options

		for i, opt in ipairs(options) do
			local Ob = Util.Instance("TextButton", List)
			Ob.Size = UDim2.new(1, 0, 0, 26)
			Ob.BackgroundColor3 = Color3.new(1, 1, 1)
			Ob.BorderSizePixel = 0
			Ob.Text = opt
			Ob.Font = Enum.Font.Code
			Ob.TextColor3 = Color3.new(1, 1, 1)
			Ob.TextSize = 16
			Ob.AutoButtonColor = false
			Ob.LayoutOrder = i
			Stylize(Ob)

			Ob.MouseEnter:Connect(function()
				Ob.BackgroundTransparency = 0.4
			end)
			Ob.MouseLeave:Connect(function()
				Ob.BackgroundTransparency = 0
			end)
			Ob.Activated:Connect(function()
				Label.Text = text .. ": " .. opt
				currentValue = opt
				pcall(callback, opt)
				isOpen = false
				Arrow.Text = "▼"
				TweenService:Create(Container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 36) }):Play()
				task.wait(0.2)
				List.Size = UDim2.new(1, margin * -2, 0, 0)
				Container.ClipsDescendants = true
			end)
			table.insert(optionButtons, Ob)
		end
	end

	refreshOptions(options)

	local function toggle()
		isOpen = not isOpen
		if isOpen then
			Container.ClipsDescendants = false
			local h = 36 + 6 + (#options * 29)
			List.Size = UDim2.new(1, margin * -2, 0, #options * 29)
			Container.Size = UDim2.new(1, 0, 0, h)
			Arrow.Text = "▲"
		else
			Arrow.Text = "▼"
			TweenService:Create(Container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 36) }):Play()
			task.wait(0.2)
			List.Size = UDim2.new(1, margin * -2, 0, 0)
			Container.ClipsDescendants = true
		end
	end

	Back.Activated:Connect(toggle)

	return {
		Refresh = refreshOptions,
		Get = function() return currentValue end,
	}
end

local Window = {}

function Window:_createUI(config)
	config = config or {}
	local title = config.Title or "UI"
	local width = config.Width or 360
	local height = config.Height or 240

	local UIMainWindow = Util.Instance("Frame", UIMainFrame)
	UIMainWindow.Active = true
	UIMainWindow.AnchorPoint = Vector2.new(0.5, 0.5)
	UIMainWindow.Position = UDim2.new(0.5, 0, 0.5, 0)
	UIMainWindow.Size = UDim2.new(0, width, 0, height)
	UIMainWindow.BackgroundColor3 = Color3.new(1, 1, 1)
	UIMainWindow.BorderSizePixel = 0
	Stylize(UIMainWindow)

	local TopBarFrame = Util.Instance("Frame", UIMainWindow)
	TopBarFrame.Position = UDim2.new(0, 0, 0, 0)
	TopBarFrame.Size = UDim2.new(1, 0, 0, 30)
	TopBarFrame.BackgroundColor3 = Color3.new(1, 1, 1)
	TopBarFrame.BorderSizePixel = 0
	TopBarFrame.ClipsDescendants = true
	TopBarFrame.ZIndex = 1
	Stylize(TopBarFrame)

	local TopBarText = Util.Instance("TextLabel", TopBarFrame)
	TopBarText.AnchorPoint = Vector2.new(0, 0.5)
	TopBarText.Position = UDim2.new(0, 8, 0.5, 0)
	TopBarText.Size = UDim2.new(1, -35, 1, 0)
	TopBarText.BackgroundTransparency = 1
	TopBarText.ClipsDescendants = true
	TopBarText.Font = Enum.Font.Code
	TopBarText.TextColor3 = Color3.new(1, 1, 1)
	TopBarText.TextSize = 20
	TopBarText.TextXAlignment = Enum.TextXAlignment.Left
	TopBarText.Text = title
	TopBarText.RichText = true
	RegisterTextLabel(TopBarText)
	Util.ForceTextSize(TopBarText)

	local TopBarClose = Util.Instance("TextButton", TopBarFrame)
	TopBarClose.AnchorPoint = Vector2.new(1, 0)
	TopBarClose.Position = UDim2.new(1, 0, 0, 0)
	TopBarClose.Size = UDim2.new(0, 30, 1, 0)
	TopBarClose.BackgroundTransparency = 1
	TopBarClose.Text = ""
	do
		local A = Util.Instance("Frame", TopBarClose)
		A.AnchorPoint = Vector2.new(0.5, 0.5)
		A.Position = UDim2.new(0.5, 0, 0.5, 0)
		A.Size = UDim2.new(0, 16, 0, 2)
		A.BackgroundColor3 = UITextColor.Value
		A.BorderSizePixel = 0
		A.Name = "A"
		A = Util.Instance("Frame", A)
		A.AnchorPoint = Vector2.new(0.5, 0.5)
		A.Position = UDim2.new(0.5, 0, 0.5, 0)
		A.Size = UDim2.new(0, 2, 0, 0)
		A.BackgroundColor3 = UITextColor.Value
		A.BorderSizePixel = 0
		A.Name = "B"
		UITextColor.Changed:Connect(function(val)
			TopBarClose.A.BackgroundColor3 = val
			TopBarClose.A.B.BackgroundColor3 = val
		end)
	end

	local WindowContent = Util.Instance("Frame", UIMainWindow)
	WindowContent.Position = UDim2.new(0, 0, 0, 30)
	WindowContent.Size = UDim2.new(1, 0, 1, -35)
	WindowContent.BackgroundTransparency = 1
	WindowContent.ClipsDescendants = true
	WindowContent.ZIndex = 0

	local Scroll = Util.Instance("ScrollingFrame", WindowContent)
	Scroll.AnchorPoint = Vector2.new(0.5, 0.5)
	Scroll.Position = UDim2.new(0.5, 0, 0.5, 0)
	Scroll.Size = UDim2.new(1, 0, 1, 0)

	Scroll.ScrollBarThickness = 3
	Scroll.ScrollBarImageColor3 = UITextColor.Value
	Scroll.ScrollBarImageTransparency = 0.5

	local Pad = Util.Instance("UIPadding", Scroll)
	Pad.PaddingTop = UDim.new(0, 5)
	Pad.PaddingBottom = UDim.new(0, 5)
	Pad.PaddingLeft = UDim.new(0, 0)
	Pad.PaddingRight = UDim.new(0, 0)

	local List = Util.Instance("UIListLayout", Scroll)
	List.FillDirection = Enum.FillDirection.Vertical
	List.HorizontalAlignment = Enum.HorizontalAlignment.Center
	List.VerticalAlignment = Enum.VerticalAlignment.Top
	List.Padding = UDim.new(0, 0)
	List.SortOrder = Enum.SortOrder.LayoutOrder

	Util.LinkDestroyI2C(Scroll, UITextColor.Changed:Connect(function(val)
		Scroll.ScrollBarImageColor3 = val
	end))
	AddToRenderStep(function(t)
		Scroll.ScrollBarImageColor3 = UITextColor.Value
	end, Scroll)

	local MainWindowClosed = false
	local MainWindowTweening = false
	local MainWindowPosOpen = nil
	local MainWindowPosClose = nil

	TopBarClose.Activated:Connect(function()
		if MainWindowTweening then return end
		MainWindowTweening = true
		MainWindowClosed = not MainWindowClosed
		if MainWindowClosed then
			MainWindowPosOpen = UIMainWindow.Position
			TweenService:Create(UIMainWindow, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Position = MainWindowPosClose or UDim2.new(0, 100, 0, 100),
				Size = UDim2.fromOffset(112, 30),
			}):Play()
			TweenService:Create(TopBarClose.A, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Rotation = 180,
			}):Play()
			TweenService:Create(TopBarClose.A.B, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Size = UDim2.new(0, 2, 0, 16),
			}):Play()
			task.delay(0.6, function()
				WindowContent.Visible = false
				MainWindowTweening = false
			end)
		else
			WindowContent.Visible = true
			MainWindowPosClose = UIMainWindow.Position
			TweenService:Create(UIMainWindow, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Position = MainWindowPosOpen,
				Size = UDim2.fromOffset(width, height),
			}):Play()
			TweenService:Create(TopBarClose.A, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Rotation = 0,
			}):Play()
			TweenService:Create(TopBarClose.A.B, TweenInfo.new(0.5, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
				Size = UDim2.new(0, 2, 0, 0),
			}):Play()
			task.delay(0.6, function()
				MainWindowTweening = false
			end)
		end
		WindowContent.Active = not MainWindowClosed
		WindowContent.Interactable = not MainWindowClosed
	end)

	local dragref = nil
	local offset = Vector2.new(0, 0)
	TopBarFrame.InputBegan:Connect(function(input)
		if dragref then return end
		if input.UserInputState ~= Enum.UserInputState.Begin then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			WindowContent.Interactable = false
			local screen = Util.GetScreenSize()
			local ch = (Vector2.new(input.Position.X, input.Position.Y) + SCREENGUI.AbsolutePosition) / screen
			offset = Util.UDim2ToVector2Scale(UIMainWindow.Position) - ch
			dragref = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragref then
			if input.UserInputType == Enum.UserInputType.MouseMovement or (input.UserInputType == Enum.UserInputType.Touch and dragref == input) then
				local screen = Util.GetScreenSize()
				local ch = (Vector2.new(input.Position.X, input.Position.Y) + SCREENGUI.AbsolutePosition) / screen
				local pos = ch + offset
				local dowsize = Util.UDim2ToVector2Offset(UIMainWindow.Size)
				local imjollx = (dowsize.X * 0.5 - 100) / screen.X
				local imjolly = dowsize.Y * -0.5 / screen.Y
				local boxy = 31 / screen.Y
				pos = Vector2.new(
					math.clamp(pos.X, -imjollx, 1 + imjollx),
					math.clamp(pos.Y, -imjolly, 1 - imjolly - boxy)
				)
				UIMainWindow.Position = Util.Vector2ToUDim2Scale(pos)
			end
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if dragref and dragref == input then
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				WindowContent.Interactable = true
				dragref = nil
			end
		end
	end)

	self.Frame = UIMainWindow
	self.Content = Scroll
	self.List = List
	self.TextColor = UITextColor
end

local Util = {}
Util.RandomString = function(n)
	n = n or 24
	local s = ""
	for _ = 1, n do s ..= string.char(math.random(32, 126)) end
	return s
end

	local i = Instance.new(c)

local function playIntro()
	local pivot = Util.Instance("Frame", MainFrame)
	pivot.AnchorPoint = Vector2.new(0.5, 0.5)
	pivot.Position = UDim2.new(0.5, 0, 0.5, 0)
	pivot.Size = UDim2.fromOffset(0, 0)
	pivot.BackgroundTransparency = 1
	pivot.ZIndex = 500

	local glow = Util.Instance("Frame", pivot)
	glow.AnchorPoint = Vector2.new(0.5, 0.5)
	glow.Position = UDim2.new(0.5, 0, 0.5, 0)
	glow.Size = UDim2.new(1, 60, 1, 60)
	glow.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	glow.BackgroundTransparency = 0.85
	glow.BorderSizePixel = 0
	Util.Instance("UICorner", glow).CornerRadius = UDim.new(1, 0)

	local circle = Util.Instance("Frame", pivot)
	circle.AnchorPoint = Vector2.new(0.5, 0.5)
	circle.Position = UDim2.new(0.5, 0, 0.5, 0)
	circle.Size = UDim2.new(1, 0, 1, 0)
	circle.BackgroundColor3 = Color3.new(1, 1, 1)
	circle.BorderSizePixel = 0
	Util.Instance("UICorner", circle).CornerRadius = UDim.new(1, 0)

	local grad = Util.Instance("UIGradient", circle)
	grad.Color = RAINBOW
	grad.Rotation = 0

	local hi = Util.Instance("Frame", circle)
	hi.AnchorPoint = Vector2.new(0.5, 0.5)
	hi.Position = UDim2.new(0.5, 0, 0.5, 0)
	hi.Size = UDim2.new(0.65, 0, 0.65, 0)
	hi.BackgroundColor3 = Color3.new(1, 1, 1)
	hi.BackgroundTransparency = 0.88
	hi.BorderSizePixel = 0
	Util.Instance("UICorner", hi).CornerRadius = UDim.new(1, 0)

	local spinConn
	spinConn = RunService.RenderStepped:Connect(function(dt)
		if not grad.Parent then spinConn:Disconnect() return end
		grad.Rotation = (grad.Rotation + dt * 100) % 360
	end)

	local vp = Util.GetScreenSize()
	local targetSize = math.floor(math.min(vp.Y * 0.32, 240))

	TweenService:Create(pivot, TweenInfo.new(1.0, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
		Size = UDim2.fromOffset(targetSize, targetSize),
	}):Play()

	task.wait(1.5)
	task.wait(1.2)

	TweenService:Create(pivot, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
		Size = UDim2.fromOffset(targetSize * 1.5, targetSize * 1.5),
	}):Play()
	TweenService:Create(circle, TweenInfo.new(0.5), { BackgroundTransparency = 1 }):Play()
	TweenService:Create(glow, TweenInfo.new(0.5), { BackgroundTransparency = 1 }):Play()

	task.wait(0.5)
	pivot:Destroy()
	spinConn:Disconnect()
end

-- 彩虹激光边框工具

local BorderUpdaters = {}

local function attachRainbowBorder(frame, thickness)
	thickness = thickness or 2
	local stroke = Util.Instance("UIStroke", frame)
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.LineJoinMode = Enum.LineJoinMode.Round
	stroke.Thickness = thickness
	stroke.Color = Color3.new(1, 1, 1)
	stroke.Transparency = 0

	local grad = Util.Instance("UIGradient", stroke)
	grad.Color = RAINBOW
	grad.Rotation = 0

	local speed = 90 + math.random(-20, 20)
	local startRot = math.random(0, 360)
	grad.Rotation = startRot

	local fn
	fn = function(dt)
		if not grad.Parent then
			local i = table.find(BorderUpdaters, fn)
			if i then table.remove(BorderUpdaters, i) end
			return
		end
		grad.Rotation = (grad.Rotation + dt * speed) % 360
	end
	table.insert(BorderUpdaters, fn)

	return stroke, grad
end

-- 每帧更新所有边框
RunService.RenderStepped:Connect(function(dt)
	for _, fn in ipairs(BorderUpdaters) do
		pcall(fn, dt)
	end
end)

local function createWindow(config)
	config = config or {}
	local title = config.Title or "Menu"
	local width = config.Width or 210
	local maxH = config.Height or 320

	local win = Util.Instance("Frame", MainFrame)
	win.AnchorPoint = Vector2.new(0, 0)
	win.Position = UDim2.new(0, 20, 0, 20)
	win.Size = UDim2.new(0, width, 0, 28)
	win.BackgroundColor3 = Color3.fromRGB(18, 18, 20)
	win.BackgroundTransparency = 0.1
	win.BorderSizePixel = 0
	win.ClipsDescendants = true
	win.Active = true

	Util.Instance("UICorner", win).CornerRadius = UDim.new(0, 8)
	attachRainbowBorder(win, 1.5)

	-- 标题栏
	local bar = Util.Instance("Frame", win)
	bar.Size = UDim2.new(1, 0, 0, 28)
	bar.BackgroundTransparency = 1

	local titleLbl = Util.Instance("TextLabel", bar)
	titleLbl.Position = UDim2.new(0, 12, 0, 0)
	titleLbl.Size = UDim2.new(1, -50, 1, 0)
	titleLbl.BackgroundTransparency = 1
	titleLbl.Font = Enum.Font.Code
	titleLbl.TextColor3 = Color3.fromRGB(240, 240, 240)
	titleLbl.TextSize = 13
	titleLbl.TextXAlignment = Enum.TextXAlignment.Left
	titleLbl.Text = title

	-- 折叠按钮
	local foldBtn = Util.Instance("TextButton", bar)
	foldBtn.AnchorPoint = Vector2.new(1, 0.5)
	foldBtn.Position = UDim2.new(1, -6, 0.5, 0)
	foldBtn.Size = UDim2.new(0, 20, 0, 20)
	foldBtn.BackgroundTransparency = 1
	foldBtn.Text = "−"
	foldBtn.Font = Enum.Font.GothamBold
	foldBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
	foldBtn.TextSize = 14
	foldBtn.AutoButtonColor = false

	-- 内容区
	local content = Util.Instance("ScrollingFrame", win)
	content.Position = UDim2.new(0, 0, 0, 28)
	content.Size = UDim2.new(1, 0, 0, 0)
	content.BackgroundTransparency = 1
	content.BorderSizePixel = 0
	content.CanvasSize = UDim2.new(0, 0, 0, 0)
	content.AutomaticCanvasSize = Enum.AutomaticSize.Y
	content.ScrollingDirection = Enum.ScrollingDirection.Y
	content.ScrollBarThickness = 2
	content.ScrollBarImageColor3 = Color3.fromRGB(80, 80, 80)
	content.ScrollBarImageTransparency = 0.3

	local pad = Util.Instance("UIPadding", content)
	pad.PaddingTop = UDim.new(0, 4)
	pad.PaddingBottom = UDim.new(0, 6)
	pad.PaddingLeft = UDim.new(0, 6)
	pad.PaddingRight = UDim.new(0, 6)

	local list = Util.Instance("UIListLayout", content)
	list.FillDirection = Enum.FillDirection.Vertical
	list.SortOrder = Enum.SortOrder.LayoutOrder
	list.Padding = UDim.new(0, 3)

	-- 自适应高度
	local function resize()
		local h = math.min(list.AbsoluteContentSize.Y + 28 + 10, maxH)
		TweenService:Create(win, TweenInfo.new(0.15), {
			Size = UDim2.new(0, width, 0, h),
		}):Play()
		content.Size = UDim2.new(1, 0, 1, -28)
	end
	list:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		if not config._folded then resize() end
	end)

	-- 折叠
	config._folded = false
	foldBtn.Activated:Connect(function()
		config._folded = not config._folded
		if config._folded then
			foldBtn.Text = "+"
			TweenService:Create(win, TweenInfo.new(0.2), {
				Size = UDim2.new(0, width, Changed0, 28),
		:Connect(function	}):Play(input()
			content.Visible = false
		else
			foldBtn.Text = "−"
			content.Visible = true
			resize()
		end
	end)

	-- 拖动
	local dragging, dragStart, startPos
	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = win.Position
		end
	end)
	UserInputService.Input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			win.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	return win, content, list, resize
end

	local dot = Util.Instance("Frame", ind)

	dot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	dot.BorderSizePixel = 0
	Util.Instance("UICorner", dot).CornerRadius = UDim.new(1, 0)

	local state = default
	local function update()
		TweenService:Create(dot, TweenInfo.new(0.15), {
			Position = state and UDim2.new(1, -12, 0.5, 0) or UDim2.new(0, 2, 0.5, 0),
		}):Play()
		TweenService:Create(ind, TweenInfo.new(0.15), {
			BackgroundColor3 = state and Color3.fromRGB(80, 200, 120) or Color3.fromRGB(60, 60, 65),
		}):Play()

	row.MouseEnter:Connect(function()
		TweenService:Create(row, TweenInfo.new(0.1), { BackgroundColor3 = Color3.fromRGB(50, 50, 55) }):Play()

	end)

	row.Activated:Connect(function()
		state = not state
		update()
		pcall(callback, state)
	end)

	return {
		Set = function(v) state = v and true or false; update(); pcall(callback, state) end,
		Get = function() return state end,
	}
end

local function makeSlider(parent, text, min, max, default, callback)
	min = min or 0
	max = max or 100
	default = default or min
	callback = callback or function() end

	local card = Util.Instance("Frame", parent)
	card.Size = UDim2.new(1, 0, 0, 38)
	card.BackgroundColor3 = Color3.fromRGB(35, 35, 38)
	card.BorderSizePixel = 0
	Util.Instance("UICorner", card).CornerRadius = UDim.new(0, 5)

	local lbl = Util.Instance("TextLabel", card)
	lbl.Position = UDim2.new(0, 10, 0, 4)
	lbl.Size = UDim2.new(1, -60, 0, 13)
	lbl.BackgroundTransparency = 1
	lbl.Font = Enum.Font.Code
	lbl.TextColor3 = Color3.fromRGB(230, 230, 230)
	lbl.TextSize = 11
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Text = text

	local valLbl = Util.Instance("TextLabel", card)
	valLbl.AnchorPoint = Vector2.new(1, 0)
	valLbl.Position = UDim2.new(1, -10, 0, 4)
	valLbl.Size = UDim2.new(0, 40, 0, 13)
	valLbl.BackgroundTransparency = 1
	valLbl.Font = Enum.Font.Code
	valLbl.TextColor3 = Color3.fromRGB(150, 210, 255)
	valLbl.TextSize = 11
	valLbl.TextXAlignment = Enum.TextXAlignment.Right
	valLbl.Text = tostring(default)

	local track = Util.Instance("Frame", card)
	track.Position = UDim2.new(0, 10, 0, 25)
	track.Size = UDim2.new(1, -20, 0, 5)
	track.BackgroundColor3 = Color3.fromRGB(20, 20, 22)
	track.BorderSizePixel = 0
	Util.Instance("UICorner", track).CornerRadius = UDim.new(1, 0)

	local pct = (default - min) / (max - min)
	local fill = Util.Instance("Frame", track)
	fill.Size = UDim2.new(pct, 0, 1, 0)
	fill.BackgroundColor3 = Color3.fromRGB(80, 180, 255)
	fill.BorderSizePixel = 0
	Util.Instance("UICorner", fill).CornerRadius = UDim.new(1, 0)

	local dot = Util.Instance("Frame", track)
	dot.AnchorPoint = Vector2.new(0.5, 0.5)
	dot.Position = UDim2.new(pct, 0, 0.5, 0)
	dot.Size = UDim2.new(0, 10, 0, 10)
	dot.BackgroundColor3 = Color3.new(1, 1, 1)
	dot.BorderSizePixel = 0
	Util.Instance("UICorner", dot).CornerRadius = UDim.new(1, 0)

	local dragging = false
	local currentVal = default
	local function setFromX(x)
		local ax = track.AbsolutePosition.X
		local aw = math.max(1, track.AbsoluteSize.X)
		local t = math.clamp((x - ax) / aw, 0, 1)
		local v = math.floor(min + (max - min) * t + 0.5)
		local p = (v - min) / (max - min)
		fill.Size = UDim2.new(p, 0, 1, 0)
		dot.Position = UDim2.new(p, 0, 0.5, 0)
		valLbl.Text = tostring(v)
		currentVal = v
		pcall(callback, v)
	end

	local hit = Util.Instance("TextButton", card)
	hit.Position = UDim2.new(0, 0, 0, 18)
	hit.Size = UDim2.new(1, 0, 0, 20)
	hit.BackgroundTransparency = 1
	hit.Text = ""
	hit.ZIndex = 5
	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	return {
		Set = function(v)
			local p = (math.clamp(v, min, max) - min) / (max - min)
			fill.Size = UDim2.new(p, 0, 1, 0)
			dot.Position = UDim2.new(p, 0, 0.5, 0)
			valLbl.Text = tostring(v)
			currentVal = v
			pcall(callback, v)
		end,
		Get = function() return currentVal end,
	}
end

		else
			arrow.Text = "v"
			TweenService:Create(wrap, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 26) }):Play()
			task.wait(0.2)
			list.Size = UDim2.new(1, 0, 0, 0)
		end
	end)

	return wrap
end

local function makeTextbox(parent, placeholder, callback)
	local wrap = Util.Instance("Frame", parent)
	wrap.Size = UDim2.new(1, 0, 0, 26)
	wrap.BackgroundColor3 = Color3.fromRGB(35, 35, 38)
	wrap.BorderSizePixel = 0
	Util.Instance("UICorner", wrap).CornerRadius = UDim.new(0, 5)

	local box = Util.Instance("TextBox", wrap)
	box.Position = UDim2.new(0, 10, 0, 0)
	box.Size = UDim2.new(1, -20, 1, 0)
	box.BackgroundTransparency = 1
	box.Font = Enum.Font.Code
	box.TextColor3 = Color3.fromRGB(230, 230, 230)
	box.TextSize = 12
	box.Text = ""
	box.PlaceholderText = placeholder or "输入..."
	box.PlaceholderColor3 = Color3.fromRGB(120, 120, 120)
	box.TextXAlignment = Enum.TextXAlignment.Left
	box.ClearTextOnFocus = false

	box.FocusLost:Connect(function(enter)
		pcall(callback, box.Text, enter)
	end)

	return {
		Get = function() return box.Text end,
		Set = function(v) box.Text = tostring(v) end,
	}
end

local function makeLabel(parent, text, size)
	local wrap = Util.Instance("Frame", parent)
	wrap.Size = UDim2.new(1, 0, 0, 20)
	wrap.BackgroundTransparency = 1

	local lbl = Util.Instance("TextLabel", wrap)
	lbl.Position = UDim2.new(0, 10, 0, 0)
	lbl.Size = UDim2.new(1, -20, 1, 0)
	lbl.BackgroundTransparency = 1
	lbl.Font = Enum.Font.Code
	lbl.TextColor3 = Color3.fromRGB(160, 160, 160)
	lbl.TextSize = size or 11
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Text = text

	return lbl
end

local function makeSeparator(parent)
	local wrap = Util.Instance("Frame", parent)
	wrap.Size = UDim2.new(1, 0, 0, 6)
	wrap.BackgroundTransparency = 1

	local line = Util.Instance("Frame", wrap)
	line.AnchorPoint = Vector2.new(0.5, 0.5)
	line.Position = UDim2.new(0.5, 0, 0.5, 0)
	line.Size = UDim2.new(1, -12, 0, 1)
	line.BackgroundColor3 = Color3.fromRGB(50, 50, 55)
	line.BorderSizePixel = 0

	return line
end

-- ============================================================
-- 窗口对象
-- ============================================================
local Window = {}
Window.__index = Window

function Window:_init(config)
	local win, content, list, resize = createWindow(config)
	self.Frame = win
	self.Content = content
	self.List = list
	self._resize = resize
end

function Window:Button(text, callback)
	return makeButton(self.Content, text, callback)
end

function Window:Toggle(text, default, callback)
	return makeToggle(self.Content, text, default, callback)
end

function Window:Slider(text, min, max, default, callback)
	return makeSlider(self.Content, text, min, max, default, callback)
end

function Window:Dropdown(text, options, callback)
	return makeDropdown(self.Content, text, options, callback)
end

function Window:Textbox(placeholder, callback)
	return makeTextbox(self.Content, placeholder, callback)
end

function Window:Label(text, size)
	return makeLabel(self.Content, text, size)
end

function Window:Separator()
	return makeSeparator(self.Content)
end

function Window:Notify(text, title)
	Util.Notify(text, title)
end

-- ============================================================
-- Library 入口
-- ============================================================
local Library = {}
Library.__index = Library

function Library.new(config)
	-- 播放开场动画
	playIntro()

	-- 等动画播完再创建窗口
	task.wait(2.7)

	local self = setmetatable({}, Library)
	local win = setmetatable({}, Window)
	win:_init(config or {})
	self.Window = win
	return win
end

-- ============================================================
return Library
